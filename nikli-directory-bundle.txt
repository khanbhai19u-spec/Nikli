package com.nikli.directory;

import android.app.Activity;
import android.content.Intent;
import android.net.Uri;
import android.os.Bundle;
import android.webkit.WebResourceRequest;
import android.webkit.WebSettings;
import android.webkit.WebView;
import android.webkit.WebViewClient;

public class MainActivity extends Activity {
    private WebView web;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        web = new WebView(this);
        setContentView(web);
        WebSettings s = web.getSettings();
        s.setJavaScriptEnabled(true);
        s.setDomStorageEnabled(true);
        s.setAllowFileAccess(true);
        s.setTextZoom(100);
        web.setWebViewClient(new WebViewClient() {
            @Override
            public boolean shouldOverrideUrlLoading(WebView v, WebResourceRequest r) {
                Uri u = r.getUrl();
                String sc = u.getScheme();
                if (sc == null || "file".equals(sc)) return false;
                try {
                    Intent i;
                    if ("tel".equals(sc)) i = new Intent(Intent.ACTION_DIAL, u);
                    else if ("mailto".equals(sc)) i = new Intent(Intent.ACTION_SENDTO, u);
                    else i = new Intent(Intent.ACTION_VIEW, u);
                    startActivity(i);
                } catch (Exception e) { }
                return true;
            }
        });
        web.loadUrl("file:///android_asset/index.html");
    }

    @Override
    public void onBackPressed() {
        web.evaluateJavascript("(window.goBack&&window.goBack())?'1':'0'", v -> {
            if (v == null || !v.contains("1")) finish();
        });
    }
}

var cat='all', query='', favs=[];
try{favs=JSON.parse(localStorage.getItem('fav')||'[]')}catch(e){}
function saveFav(){try{localStorage.setItem('fav',JSON.stringify(favs))}catch(e){}}
function esc(s){return String(s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;')}
function tel(n){return 'tel:'+n.replace(/[^0-9+]/g,'')}
var toastT;
function toast(m){var t=document.getElementById('toast');t.textContent=m;t.className='show';clearTimeout(toastT);toastT=setTimeout(function(){t.className=''},1600)}
function copy(txt){
  function fb(){var a=document.createElement('textarea');a.value=txt;a.style.position='fixed';a.style.opacity='0';document.body.appendChild(a);a.select();try{document.execCommand('copy')}catch(e){}document.body.removeChild(a)}
  try{if(navigator.clipboard&&navigator.clipboard.writeText){navigator.clipboard.writeText(txt).catch(fb)}else fb()}catch(e){fb()}
  toast('কপি হয়েছে: '+txt);
}

function card(it){
  var n=it[0],ph=it[1],em=it[2],wb=it[3],on=favs.indexOf(n)>-1;
  var h='<div class="card"><div class="top"><div class="nm">'+esc(n)+'</div><button class="star'+(on?' on':'')+'" data-fav="'+esc(n)+'" aria-label="প্রিয়">'+IC.star+'</button></div><div class="row">';
  ph.forEach(function(p){
    h+='<a class="btn call" href="'+tel(p)+'">'+IC.call+p+'</a>';
    h+='<button class="btn sm" data-copy="'+p+'" aria-label="কপি">'+IC.copy+'</button>';
  });
  if(em)h+='<a class="btn sm" href="mailto:'+esc(em)+'">'+IC.mail+esc(em)+'</a>';
  if(wb)h+='<a class="btn sm" href="https://'+esc(wb)+'">'+IC.web+esc(wb)+'</a>';
  return h+'</div></div>';
}

function render(){
  var q=query.trim().toLowerCase(), out='', found=0;
  if(cat==='all'&&!q){
    out+='<div class="sos"><h3>জরুরি প্রয়োজনে</h3><div class="sosrow">';
    SOS.forEach(function(s){out+='<a href="'+tel(s[0])+'"><b>'+s[0]+'</b><span>'+s[1]+'</span></a>'});
    out+='</div></div>';
  }
  out+='<div class="chips"><button class="chip'+(cat==='all'?' on':'')+'" data-cat="all">সব</button><button class="chip'+(cat==='fav'?' on':'')+'" data-cat="fav">প্রিয়</button>';
  D.forEach(function(s){out+='<button class="chip'+(cat===s.id?' on':'')+'" data-cat="'+s.id+'">'+s.t+'</button>'});
  out+='</div>';
  D.forEach(function(s){
    if(cat!=='all'&&cat!=='fav'&&cat!==s.id)return;
    var items=s.i.filter(function(it){
      if(cat==='fav'&&favs.indexOf(it[0])<0)return false;
      if(!q)return true;
      return (it[0]+' '+it[1].join(' ')+' '+(it[2]||'')+' '+(it[3]||'')+' '+s.t).toLowerCase().indexOf(q)>-1;
    });
    if(!items.length)return;
    found+=items.length;
    out+='<div class="sec">'+s.t+'</div>';
    items.forEach(function(it){out+=card(it)});
  });
  if(!found)out+='<div class="empty">'+(cat==='fav'&&!q?'কোনো প্রিয় নম্বর নেই। কার্ডের তারা চিহ্নে চাপ দিয়ে যোগ করুন।':'কিছু পাওয়া যায়নি')+'</div>';
  out+='<div class="foot">নিকলী, কিশোরগঞ্জ</div>';
  document.getElementById('main').innerHTML=out;
}

document.addEventListener('click',function(e){
  var b=e.target.closest('[data-cat],[data-copy],[data-fav]');
  if(!b)return;
  var m=document.getElementById('main'), top=m.scrollTop;
  if(b.dataset.cat){cat=b.dataset.cat;render();m.scrollTop=0}
  else if(b.dataset.copy){copy(b.dataset.copy)}
  else if(b.dataset.fav){
    var n=b.dataset.fav,k=favs.indexOf(n);
    if(k>-1)favs.splice(k,1);else favs.push(n);
    saveFav();render();m.scrollTop=top;
    toast(k>-1?'প্রিয় থেকে সরানো হয়েছে':'প্রিয়তে যোগ হয়েছে');
  }
});

  if(query){qi.value='';query='';cl.style.display='none';render();return true}
  if(cat!=='all'){cat='all';render();return true}
  return false;
};
